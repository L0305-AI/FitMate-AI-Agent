# ==========================================
# 1. Web 框架与 ASGI 服务器
# ==========================================
fastapi               # 核心 Web 后端框架，提供 RESTful API
uvicorn               # 轻量级 ASGI 服务器，用于运行 FastAPI 应用
pydantic              # 数据验证库，用于定义请求和响应的 JSON 数据模型

# ==========================================
# 2. 大模型接入与 Agent 状态图编排
# ==========================================
langchain             # LangChain 核心库，用于大模型交互与基础组件
langchain-community   # LangChain 社区扩展包，提供文档加载器等工具
langchain-openai      # 兼容 OpenAI 接口的模型接入层，用于连接通义千问 API
langgraph             # LangGraph 状态图库，用于构建 Agent 的自主规划与条件路由

# ==========================================
# 3. RAG 向量检索与文本处理
# ==========================================
qdrant-client         # Qdrant 向量数据库的 Python 客户端，用于存储和检索知识库
tiktoken              # 文本 Token 计算库，辅助大模型估算上下文长度

# ==========================================
# 4. 关系型数据库与 ORM 映射
# ==========================================
sqlalchemy            # 强大的 Python ORM 框架，用于操作 MySQL 数据库
pymysql               # MySQL 数据库驱动，使 SQLAlchemy 能连接 MySQL
cryptography          # 密码学工具包，PyMySQL 连接 MySQL 8.0 时的必需安全依赖

# ==========================================
# 5. 环境变量与工具库
# ==========================================
python-dotenv         # 读取 .env 文件中的环境变量（如 API Key、数据库密码）
